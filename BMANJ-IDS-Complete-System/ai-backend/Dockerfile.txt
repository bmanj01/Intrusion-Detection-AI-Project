# Base Python image
FROM python:3.11-slim

# Set working directory inside container
WORKDIR /app

# Install system dependencies (needed for numpy/pandas/sklearn)
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    && rm -rf /var/lib/apt/lists/*

# Copy requirements and install Python deps
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy API code and model into the image
COPY serve_model.py ./serve_model.py
COPY artifacts/model_tuned.pkl ./artifacts/model_tuned.pkl

# Expose the API port
EXPOSE 8000

# Start FastAPI with uvicorn
CMD ["uvicorn", "serve_model:app", "--host", "0.0.0.0", "--port", "8000"]
